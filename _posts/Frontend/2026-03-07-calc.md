    <script>
        /**
         * =========================================
         * JavaScript 逻辑部分
         * 包含两个主要部分：
         * 1. Calculator Class: 处理 UI 交互和状态管理
         * 2. ExpressionParser: 安全的数学表达式计算引擎
         * =========================================
         */

        // --- 部分 1: 安全的数学表达式解析器 ---
        // 为了避免使用 eval() 带来的安全风险，我们手动实现一个简单的解析器
        // 支持：+ - * / 和 括号 () 优先级
        
        class ExpressionParser {
            constructor(expression) {
                this.expression = expression.replace(/×/g, '*').replace(/÷/g, '/');
                this.index = 0;
            }

            calculate() {
                try {
                    const result = this.parseExpression();
                    if (this.index < this.expression.length) {
                        throw new Error("无效字符");
                    }
                    return result;
                } catch (e) {
                    return "Error";
                }
            }

            // 处理加减法 (最低优先级)
            parseExpression() {
                let left = this.parseTerm();
                
                while (this.index < this.expression.length) {
                    const char = this.expression[this.index];
                    if (char === '+') {
                        this.index++;
                        left += this.parseTerm();
                    } else if (char === '-') {
                        this.index++;
                        left -= this.parseTerm();
                    } else {
                        break;
                    }
                }
                return left;
            }

            // 处理乘除法 (高优先级)
            parseTerm() {
                let left = this.parseFactor();
                
                while (this.index < this.expression.length) {
                    const char = this.expression[this.index];
                    if (char === '*') {
                        this.index++;
                        left *= this.parseFactor();
                    } else if (char === '/') {
                        this.index++;
                        const right = this.parseFactor();
                        if (right === 0) throw new Error("除零错误");
                        left /= right;
                    } else {
                        break;
                    }
                }
                return left;
            }

            // 处理数字和括号 (最高优先级)
            parseFactor() {
                if (this.index >= this.expression.length) return 0;

                const char = this.expression[this.index];

                // 处理括号
                if (char === '(') {
                    this.index++; // 跳过 '('
                    const result = this.parseExpression();
                    if (this.expression[this.index] !== ')') {
                        throw new Error("缺少右括号");
                    }
                    this.index++; // 跳过 ')'
                    return result;
                }

                // 处理负数 (例如 -5)
                if (char === '-') {
                    // 简单判断：如果是开头或者前一个是运算符，则视为负号
                    const prevChar = this.index > 0 ? this.expression[this.index - 1] : '';
                    if (this.index === 0 || ['+', '-', '*', '/', '('].includes(prevChar)) {
                        this.index++;
                        return -this.parseFactor();
                    }
                }

                // 处理数字
                return this.parseNumber();
            }

            parseNumber() {
                let numStr = '';
                let hasDot = false;

                while (this.index < this.expression.length) {
                    const char = this.expression[this.index];
                    if ((char >= '0' && char <= '9') || (char === '.' && !hasDot)) {
                        if (char === '.') hasDot = true;
                        numStr += char;
                        this.index++;
                    } else {
                        break;
                    }
                }

                if (numStr === '' || numStr === '.') throw new Error("无效数字");
                return parseFloat(numStr);
            }
        }

        // --- 部分 2: 计算器控制器类 ---

        class Calculator {
            constructor(previousOperandTextElement, currentOperandTextElement) {
                this.previousOperandTextElement = previousOperandTextElement;
                this.currentOperandTextElement = currentOperandTextElement;
                this.readyToReset = false; // 标记是否刚计算完，准备输入新数字
                this.clear();
            }

            // 清空所有
            clear() {
                this.currentOperand = '0';
                this.previousOperand = '';
                this.operation = undefined;
                this.readyToReset = false;
                this.updateDisplay();
            }

            // 删除最后一位
            delete() {
                if (this.readyToReset) {
                    this.clear();
                    return;
                }
                if (this.currentOperand === '0') return;
                if (this.currentOperand.length === 1) {
                    this.currentOperand = '0';
                } else {
                    this.currentOperand = this.currentOperand.toString().slice(0, -1);
                }
                this.updateDisplay();
            }

            // 添加数字
            appendNumber(number) {
                // 如果刚计算完，输入数字则开始新的计算
                if (this.readyToReset) {
                    this.currentOperand = '';
                    this.readyToReset = false;
                    this.previousOperand = '';
                }

                // 防止多个小数点
                if (number === '.' && this.currentOperand.includes('.')) return;
                
                // 防止多个前导零 (除非是小数)
                if (this.currentOperand === '0' && number !== '.') {
                    this.currentOperand = number.toString();
                } else {
                    this.currentOperand = this.currentOperand.toString() + number.toString();
                }
                this.updateDisplay();
            }

            // 选择运算符
            chooseOperation(operation) {
                if (this.currentOperand === '') return;
                
                // 如果已经有运算符且未输入新数字，则切换运算符
                if (this.previousOperand !== '' && this.readyToReset) {
                    this.operation = operation;
                    this.updateDisplay();
                    return;
                }

                if (this.previousOperand !== '') {
                    this.compute();
                }
                
                this.operation = operation;
                this.previousOperand = this.currentOperand;
                this.currentOperand = '';
                this.updateDisplay();
            }

            // 执行计算
            compute() {
                let computation;
                
                // 构建完整的表达式字符串用于解析
                // 注意：这里我们利用之前存储的状态构建字符串，而不是直接操作 DOM
                const prev = this.previousOperand;
                const current = this.currentOperand;
                const op = this.operation;

                if (!op || !current) return;

                // 使用自定义的安全解析器
                const expressionString = `${prev}${op}${current}`;
                const parser = new ExpressionParser(expressionString);
                const result = parser.calculate();

                if (result === "Error") {
                    this.currentOperand = "Error";
                    this.previousOperand = '';
                    this.operation = undefined;
                } else {
                    // 处理浮点数精度问题 (例如 0.1 + 0.2)
                    this.currentOperand = Math.round(result * 100000000) / 100000000;
                    this.previousOperand = '';
                    this.operation = undefined;
                }
                
                this.readyToReset = true;
                this.updateDisplay();
            }

            // 格式化数字显示 (添加千分位逗号)
            getDisplayNumber(number) {
                const stringNumber = number.toString();
                const integerDigits = parseFloat(stringNumber.split('.')[0]);
                const decimalDigits = stringNumber.split('.')[1];
                
                let integerDisplay;
                if (isNaN(integerDigits)) {
                    integerDisplay = '';
                } else {
                    integerDisplay = integerDigits.toLocaleString('en', { maximumFractionDigits: 0 });
                }

                if (decimalDigits != null) {
                    return `${integerDisplay}.${decimalDigits}`;
                } else {
                    return integerDisplay;
                }
            }

            // 更新界面
            updateDisplay() {
                this.currentOperandTextElement.innerText = this.getDisplayNumber(this.currentOperand);
                
                if (this.operation != null) {
                    this.previousOperandTextElement.innerText = 
                        `${this.getDisplayNumber(this.previousOperand)} ${this.operation}`;
                } else {
                    this.previousOperandTextElement.innerText = '';
                }
            }
        }

        // --- 初始化 ---
        const previousOperandTextElement = document.getElementById('previous-operand');
        const currentOperandTextElement = document.getElementById('current-operand');

        const calculator = new Calculator(previousOperandTextElement, currentOperandTextElement);

        // 添加键盘支持
        document.addEventListener('keydown', (event) => {
            const key = event.key;
            
            if ((key >= '0' && key <= '9') || key === '.') {
                calculator.appendNumber(key);
            } else if (key === '+' || key === '-' || key === '*' || key === '/') {
                // 映射键盘符号到显示符号
                let op = key;
                if (key === '*') op = '×';
                if (key === '/') op = '÷';
                calculator.chooseOperation(op);
            } else if (key === 'Enter' || key === '=') {
                event.preventDefault(); // 防止回车提交表单
                calculator.compute();
            } else if (key === 'Backspace') {
                calculator.delete();
            } else if (key === 'Escape' || key === 'c' || key === 'C') {
                calculator.clear();
            }
        });

    </script>
